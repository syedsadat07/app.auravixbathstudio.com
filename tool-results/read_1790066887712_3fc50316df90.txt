   297→  async function buildVersionData(opts: {
   298→    customerName: string
   299→    bathrooms: Array<{ name: string; type: string; length: number; width: number; height?: number }>
   300→    packageName: string
   301→    tileName: string
   302→    fixtures: Array<{ material: string; qty: number }>
   303→    addons: Array<{ material: string; qty: number }>
   304→    consultation: boolean
   305→    discountType: 'FLAT' | 'PERCENT'
   306→    discount: number
   307→  }) {
   308→    const pkg = await db.package.findUnique({ where: { name: opts.packageName } })
   309→    if (!pkg) throw new Error('package missing')
   310→    const bathrooms = opts.bathrooms.map((b) => ({ ...b, areaSqft: Math.round(b.length * b.width * 100) / 100 }))
   311→    const totalArea = bathrooms.reduce((s, b) => s + b.areaSqft, 0)
   312→    const packageAmount = totalArea * pkg.pricePerSqft
   313→
   314→    const items: Array<{ section: string; materialId: string | null; name: string; quantity: number; unit: string; rate: number; amount: number }> = []
   315→    const tile = await db.material.findUnique({ where: { id: materialIdByName[opts.tileName] } })
   316→    if (tile) {
   317→      const qty = Math.ceil((totalArea / tile.coverage) * (1 + tile.wastagePercent / 100))
   318→      items.push({ section: 'TILES', materialId: tile.id, name: tile.name, quantity: qty, unit: tile.unit, rate: tile.price, amount: qty * tile.price })
   319→    }
   320→    for (const f of opts.fixtures) {
   321→      const m = await db.material.findUnique({ where: { id: materialIdByName[f.material] } })
   322→      if (m) items.push({ section: 'FIXTURES', materialId: m.id, name: m.name, quantity: f.qty, unit: m.unit, rate: m.price, amount: f.qty * m.price })
   323→    }
   324→    for (const a of opts.addons) {
   325→      const m = await db.material.findUnique({ where: { id: materialIdByName[a.material] } })
   326→      if (m) items.push({ section: 'ADDONS', materialId: m.id, name: m.name, quantity: a.qty, unit: m.unit, rate: m.price, amount: a.qty * m.price })
   327→    }
   328→    if (opts.consultation) {
   329→      items.push({ section: 'OTHER', materialId: null, name: 'Design Consultation', quantity: 1, unit: 'unit', rate: 299, amount: 299 })
   330→    }
   331→    const itemsAmount = items.reduce((s, i) => s + i.amount, 0)
   332→    const grossValue = packageAmount + itemsAmount
   333→    const discountAmount = opts.discountType === 'PERCENT' ? (grossValue * opts.discount) / 100 : opts.discount
   334→    const taxableValue = grossValue - discountAmount
   335→    const gstAmount = (taxableValue * gstRate) / 100
   336→    const totalPayable = taxableValue + gstAmount
   337→    return { bathrooms, totalArea, packageId: pkg.id, packageName: pkg.name, packageRate: pkg.pricePerSqft, packageAmount, items, itemsAmount, grossValue, discountType: opts.discountType, discount: opts.discount, discountAmount, taxableValue, gstRate, gstAmount, totalPayable }
   338→  }
